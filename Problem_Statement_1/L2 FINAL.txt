void Normal(int R, int C , int G)
{
    cout<<"\n* = Dead , # = Alive \nEnter the characters * and # for each row one by one\n";
    char Grid[R][C];
    int InitialPopulation = 0;
    
    char Temp[R][C];
    char Temp1[R][C];

    
    for(int i=0; i<R ; i++)
    {
        cout<<"Row "<<i+1<<endl;
        for(int j=0; j<C ; j++)
        {
            cin>>Grid[i][j];
            if(Grid[i][j] == '#')
            {
                InitialPopulation++;
            }
        }
    }
    
    memcpy(Temp , Grid , sizeof(Grid));
    
    int MaxPopulation = InitialPopulation;
    
    for(int i=0 ; i<G ;i++)
    {
        for(int i =0; i<R ; i++)
        {
            for(int j=0; j<C ; j++)
            {
                int Alive = 0;
                int Dead = 0;
                
                if(i-1<0 || Temp[i-1][j]=='*')                                  //1
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }  
                
                if(i+1>=R || Temp[i+1][j]=='*')                                 //2
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }
                
                if(j-1<0 || Temp[i][j-1]=='*')                                  //3
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }
                
                if(j+1>=C || Temp[i][j+1]=='*')                                 //4
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }
                 
                if(i-1<0 || j-1<0 || Temp[i-1][j-1]=='*')                       //5
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }
                
                if(i-1<0 || j+1>=C || Temp[i-1][j+1]=='*')                      //6
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }
                
                if(i+1>=R || j-1<0 || Temp[i+1][j-1]=='*')                      //7
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }
                
                if(i+1>=R || j+1>=C || Temp[i+1][j+1]=='*')                      //8
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }
                
                
                if(Temp[i][j] == '#')
                    {
                        if(Alive<2)
                        {
                            Temp1[i][j] = '*';
                        }
                        else if(Alive>3)
                        {
                            Temp1[i][j] = '*';
                        }
                        else
                        {
                            Temp1[i][j] = '#';
                        }
                    }
                else
                {
                    if(Alive == 3)
                    {
                        Temp1[i][j] = '#';
                    }
                    else
                    {
                        Temp1[i][j] = '*';
                    }
                }
                
            }
        }
        int TempPopulation = 0;
        for(int i=0; i<R ; i++)
        {
            for(int j=0; j<C ; j++)
            {
                if(Temp1[i][j] == '#')
                {
                    TempPopulation++;
                }
            }
        }    
        if(TempPopulation>MaxPopulation)
        {
            MaxPopulation = TempPopulation;
        }
        
        memcpy(Temp , Temp1 , sizeof(Temp1));
    }        

    int FinalPopulation = 0;

    for(int i=0; i<R ; i++)
    {
        for(int j=0; j<C ; j++)
        {
            if(Temp[i][j] == '#')
            {
                FinalPopulation++;
            }
        }
    }
    
    cout<<"Initial Population = "<<InitialPopulation<<endl;
    cout<<"Final Population = "<<FinalPopulation<<endl;
    cout<<"Maximum Population = "<<MaxPopulation<<endl;
    cout<<endl;
    
    for(int i=0; i<R ; i++)
    {
        for(int j=0; j<C ; j++)
        {    
            cout<<Temp[i][j]<<" ";
        }
        cout<<endl;
    }    
}

void PatternClassifier(int R, int C , int G)
{
    char GenData[G + 1][R][C];

    cout<<"\n* = Dead , # = Alive \nEnter the characters * and # for each row one by one\n";

    char Grid[R][C];
    int InitialPopulation = 0;

    int GenPopulation[G];
    int ExtinctionStep;

    char Temp[R][C];
    char Temp1[R][C];


    for(int i=0; i<R ; i++)
    {
        cout<<"Row "<<i+1<<endl;

        for(int j=0; j<C ; j++)
        {
            cin>>Grid[i][j];

            if(Grid[i][j] == '#')
            {
                InitialPopulation++;
            }
        }
    }

    memcpy(GenData[0], Grid, sizeof(Grid));
    memcpy(Temp , Grid , sizeof(Grid));

    if(InitialPopulation == 0)
    {
        cout<<"Classification: Extinct\n";
        cout<<"Extinction Step: 0\n";
        cout<<"Final Population: 0\n";
        return;
    }


    int FinalPopulation = InitialPopulation;
    bool extinct = false;


    for(int m=0 ; m<G ;m++)
    {
        for(int i =0; i<R ; i++)
        {
            for(int j=0; j<C ; j++)
            {
                int Alive = 0;
                int Dead = 0;


                if(i-1<0 || Temp[i-1][j]=='*')       //1
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }


                if(i+1>=R || Temp[i+1][j]=='*')      //2
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }


                if(j-1<0 || Temp[i][j-1]=='*')       //3
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }


                if(j+1>=C || Temp[i][j+1]=='*')      //4
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }


                if(i-1<0 || j-1<0 || Temp[i-1][j-1]=='*')   //5
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }


                if(i-1<0 || j+1>=C || Temp[i-1][j+1]=='*')  //6
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }


                if(i+1>=R || j-1<0 || Temp[i+1][j-1]=='*')  //7
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }


                if(i+1>=R || j+1>=C || Temp[i+1][j+1]=='*') //8
                {
                    Dead++;
                }
                else
                {
                    Alive++;
                }

                if(Temp[i][j] == '#')
                {
                    if(Alive<2)
                    {
                        Temp1[i][j] = '*';
                    }
                    else if(Alive>3)
                    {
                        Temp1[i][j] = '*';
                    }
                    else
                    {
                        Temp1[i][j] = '#';
                    }
                }
                else
                {
                    if(Alive == 3)
                    {
                        Temp1[i][j] = '#';
                    }
                    else
                    {
                        Temp1[i][j] = '*';
                    }
                }
            }
        }


        int TempPopulation = 0;

        for(int i=0; i<R ; i++)
        {
            for(int j=0; j<C ; j++)
            {
                if(Temp1[i][j] == '#')
                {
                    TempPopulation++;
                }
            }
        }


        GenPopulation[m] = TempPopulation;

        memcpy(GenData[m + 1], Temp1, sizeof(Temp1));
        memcpy(Temp , Temp1 , sizeof(Temp1));


        FinalPopulation = TempPopulation;

        if(TempPopulation == 0)
        {
            ExtinctionStep = m + 1;
            extinct = true;
            break;
        }
    }

    if(extinct)
    {
        cout<<"Classification: Extinct\n";
        cout<<"Extinction Step: "<<ExtinctionStep<<endl;
        cout<<"Final Population: 0\n";
        return;
    }

    bool found = false;

    for(int i=1; i<=G && !found ; i++)
    {
        for(int j=0; j<i ; j++)
        {
            if(memcmp(GenData[i], GenData[j],
                      sizeof(GenData[0])) == 0)
            {
                int period = i-j;


                if(period != 1)
                {
                    cout<<"Classification: Oscillator\n";
                    cout<<"Period: "<<period<<endl;
                    cout<<"First Repeat Step: "<<i
                        <<" (matches Step "<<j<<")"<<endl;
                    cout<<"Final Population: "
                        <<FinalPopulation<<endl;

                    found = true;
                    break;
                }


                else if(period == 1)
                {
                    cout<<"Classification: Still Life\n";
                    cout<<"Stable at Step: "<<j<<endl;
                    cout<<"Period: 1\n";
                    cout<<"Final Population: "
                        <<FinalPopulation<<endl;

                    found = true;
                    break;
                }
            }
        }
    }

    if(!found)
    {
        cout<<"Classification: Active\n";
        cout<<"Reason: No repeat or extinction detected within K = "
            <<G<<" steps\n";
        cout<<"Final Population: "
            <<FinalPopulation<<endl;
    }
}
